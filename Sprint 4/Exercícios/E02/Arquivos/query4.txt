WITH ranking_nome AS (
    SELECT 
        nome, 
        SUM(total) AS total_nomes,
        FLOOR(ano / 10) * 10 AS decada,
        RANK() OVER (PARTITION BY FLOOR(ano / 10) * 10 ORDER BY SUM(total) DESC) AS ranking
    FROM tb_consulta
    WHERE ano >= 1950
    GROUP BY nome, FLOOR(ano / 10) * 10
)
SELECT nome, total_nomes, decada
FROM ranking_nome
WHERE ranking <= 3
ORDER BY decada, ranking
